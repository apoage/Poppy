## 2. PROPOSED SOLUTION: Single Batched Call

### Key Insight from jax_encoder.py

`jit_batch_encode_simple` already supports `per_env=True`:

```python
@partial(jax.jit, static_argnames=['config', 'per_env'])
def jit_batch_encode_simple(prices, window_starts, config, per_env=False):
    if per_env:
        # prices: (n_envs, n_steps, n_assets)
        # window_starts: (n_envs,)
        return jax.vmap(
            lambda p, ws: encode_observation_simple(p, ws, config)
        )(prices, window_starts)
```

### Challenge: Variable Episode Lengths

Episodes have different lengths:
- Episode A: 400 steps (prices.shape = (400, 64))
- Episode B: 350 steps (prices.shape = (350, 64))
- Episode C: 420 steps (prices.shape = (420, 64))

JAX requires static shapes for `vmap` with `per_env=True`.

### Solution: Pad to Max Length

```python
# Find max episode length among sampled episodes
```

```python
def _encode_observations_batched(self, entries):
    """Encode all observations in single JAX call."""
    import jax
    import jax.numpy as jnp
    from mcts_trader.env.jax_encoder import EncoderConfig, jit_batch_encode_simple
    
    episode_ids = entries['episode_id'].astype(np.int32)
    unique_episodes = np.unique(episode_ids)
    n_episodes = len(unique_episodes)
    n_assets = self.n_assets
    
    # Build padded price array: (n_episodes, max_len, n_assets)
    max_len = max(self._episode_prices[ep_id].shape[0] 
                  for ep_id in unique_episodes)
    
    padded_prices = np.zeros((n_episodes, max_len, n_assets), dtype=np.float32)
    actual_lengths = np.zeros(n_episodes, dtype=np.int32)
    
    for i, ep_id in enumerate(unique_episodes):
        prices = self._episode_prices[ep_id]
        actual_lengths[i] = len(prices)
        padded_prices[i, :len(prices)] = prices
    
    # Build per-episode window_starts list
    episode_window_starts = []
    max_samples = 0
    
    for ep_id in unique_episodes:
        mask = episode_ids == ep_id
        steps = entries['step'][mask].astype(np.int32)
        # Filter valid steps
        valid_mask = steps >= self.window_size
        steps = steps[valid_mask]
        window_starts = steps - self.window_size
        episode_window_starts.append(window_starts)
        max_samples = max(max_samples, len(window_starts))
    
    # Pad window_starts to (n_episodes, max_samples)
    padded_window_starts = np.full((n_episodes, max_samples), -1, dtype=np.int32)
    sample_counts = np.zeros(n_episodes, dtype=np.int32)
    
    for i, starts in enumerate(episode_window_starts):
        padded_window_starts[i, :len(starts)] = starts
        sample_counts[i] = len(starts)
    
    # Single JAX call!
    config = EncoderConfig(n_assets=n_assets, window_size=self.window_size)
    
    # Encode: (n_episodes, max_samples, 27, n_assets, n_assets)
    # Note: jit_batch_encode_simple with per_env=True takes (n_envs, n_steps, n_assets)
    # But we need vmapped over window_starts too...
    
    # Better: use jax.vmap at two levels
    @jax.jit
    def encode_episode_batch(prices_episode, window_starts_episode):
        # prices_episode: (max_len, n_assets)
        # window_starts_episode: (max_samples,)
        return jax.vmap(
            lambda ws: encode_observation_simple(prices_episode, ws, config)
        )(window_starts_episode)
    
    # Vmap over episodes
    encode_all = jax.vmap(encode_episode_batch)
    
    # Single call!
    all_encoded = encode_all(
        jnp.array(padded_prices), 
        jnp.array(padded_window_starts)
    )  # (n_episodes, max_samples, 27, n_assets, n_assets)
    
    # Scatter back to output (handling padding)
    observations = np.zeros((len(entries), 27, n_assets, n_assets), dtype=np.float32)
    
    idx = 0
    for i, ep_id in enumerate(unique_episodes):
        n_samples = sample_counts[i]
        encoded = np.array(all_encoded[i, :n_samples])
        
        # Map to original batch indices
        mask = episode_ids == ep_id
        batch_indices = np.where(mask)[0]
        # Filter to valid steps only
        steps = entries['step'][mask].astype(np.int32)
        valid_mask = steps >= self.window_size
        batch_indices = batch_indices[valid_mask]
        
        observations[batch_indices] = encoded
        idx += n_samples
    
    return observations
```

## 5. IMPLEMENTATION PLAN

### Option A: Two-Level Vmap (Recommended)

```python
def _encode_observations_vectorized(self, entries):
    """Vectorized observation encoding - single JAX call."""
    import jax
    import jax.numpy as jnp
    from functools import partial
    from mcts_trader.env.jax_encoder import EncoderConfig, encode_observation_simple
    
    batch_size = len(entries)
    n_assets = self.n_assets
    
    # Group by episode
    episode_ids = entries['episode_id'].astype(np.int32)
    unique_episodes = np.unique(episode_ids)
    n_episodes = len(unique_episodes)
    
    if n_episodes == 0:
        return np.zeros((batch_size, 27, n_assets, n_assets), dtype=np.float32)
    
    # Build padded price array
    max_len = max(self._episode_prices[ep_id].shape[0] 
                  for ep_id in unique_episodes)
    padded_prices = np.zeros((n_episodes, max_len, n_assets), dtype=np.float32)
    
    for i, ep_id in enumerate(unique_episodes):
        prices = self._episode_prices[ep_id]
        padded_prices[i, :len(prices)] = prices
    
    # Build per-episode window_starts
    episode_window_starts = []
    max_samples = 0
    
    for ep_id in unique_episodes:
        mask = episode_ids == ep_id
        steps = entries['step'][mask].astype(np.int32)
        # Filter to valid steps
        valid_mask = steps >= self.window_size
        steps = steps[valid_mask]
        window_starts = (steps - self.window_size).astype(np.int32)
        episode_window_starts.append(window_starts)
        max_samples = max(max_samples, len(window_starts))
    
    # Pad window_starts
    padded_window_starts = np.full((n_episodes, max_samples), -1, dtype=np.int32)
    sample_counts = np.zeros(n_episodes, dtype=np.int32)
    
    for i, starts in enumerate(episode_window_starts):
        padded_window_starts[i, :len(starts)] = starts
        sample_counts[i] = len(starts)
    
    # Config
    config = EncoderConfig(n_assets=n_assets, window_size=self.window_size)
    
    # Two-level vmap - single JIT compilation
    @partial(jax.jit, static_argnames=['config'])
    def encode_all(prices_batch, starts_batch):
        # prices_batch: (n_episodes, max_len, n_assets)
        # starts_batch: (n_episodes, max_samples)
        
        def encode_one_episode(prices, starts):
            # prices: (max_len, n_assets)
            # starts: (max_samples,)
            
            # Mask invalid starts (-1)
            valid_mask = starts >= 0
            
            # Use lax.map or vmap with masking
            def encode_or_zero(start):
                return jax.lax.cond(
                    start >= 0,
                    lambda s: encode_observation_simple(prices, s, config),
                    lambda s: jnp.zeros((27, n_assets, n_assets)),
                    start
                )
            
            return jax.vmap(encode_or_zero)(starts)
        
        return jax.vmap(encode_one_episode)(prices_batch, starts_batch)
    
    # Single call
    all_encoded = encode_all(
        jnp.array(padded_prices),
        jnp.array(padded_window_starts)
    )  # (n_episodes, max_samples, 27, n_assets, n_assets)
    
    # Extract valid results
    observations = np.zeros((batch_size, 27, n_assets, n_assets), dtype=np.float32)
    
    out_idx = 0
    for i, ep_id in enumerate(unique_episodes):
        n_samples = sample_counts[i]
        if n_samples == 0:
            continue
            
        # Get encoded for this episode
        encoded_ep = np.array(all_encoded[i, :n_samples])
        
        # Map to original batch indices
        mask = episode_ids == ep_id
        batch_indices = np.where(mask)[0]
        steps = entries['step'][mask].astype(np.int32)
        valid_mask = steps >= self.window_size
        batch_indices = batch_indices[valid_mask]
        
        if len(batch_indices) == n_samples:
            observations[batch_indices] = encoded_ep
        
        out_idx += n_samples
    
    return observations
```

### Option C: Padded Batching with jax.lax.dynamic_slice

Use dynamic slicing to avoid per-episode vmapping:

```python
@jax.jit
def encode_from_padded(padded_prices, window_starts, episode_indices):
    # episode_indices: which episode each sample belongs to
    # window_starts: absolute window start for each sample
    
    def encode_one(sample_idx):
        ep_idx = episode_indices[sample_idx]
        ws = window_starts[sample_idx]
        prices = padded_prices[ep_idx]  # (max_len, n_assets)
        return encode_observation_simple(prices, ws, config)
    
    return jax.vmap(encode_one)(jnp.arange(len(window_starts)))
```

This avoids the two-level vmap complexity!
