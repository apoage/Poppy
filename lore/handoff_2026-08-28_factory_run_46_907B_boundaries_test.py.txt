def _dense_capture(path, pairs_data, span_s=600.0, cadence_s=5.0, channels=("book", "trade"),
                    declared_pairs=None, t0=T0):
    """A capture built through the PRODUCTION writer (`CaptureWriter`), one session, every pair
    in `pairs_data` carrying dense book+trade for `span_s`. `pairs_data` maps pair -> price."""
    declared = declared_pairs if declared_pairs is not None else sorted(pairs_data)
    w = CaptureWriter(str(path), declared, "cap", channels=list(channels) if channels else None)
    t = 0.0
    while t <= span_s:
        for pair, px in pairs_data.items():
            w.write_book(pair, t0 + t, px, px + 0.1, 1.0, 1.0)
            w.write_trade(pair, t0 + t + 0.01, px + 0.05, 1.0, "buy")
        t += cadence_s
    w.close(clean=True)
    return str(path)

class _FastNow:
    """Swap `capture._now` for a fixed value, restoring it on exit — used to control the resume
    timestamp deterministically (mirrors `test_capture_gaps.py::test_resume_writes_gap...`)."""

    def __init__(self, ts):
        self.ts = ts

    def __enter__(self):
        import capture as capture_mod
        self._mod = capture_mod
        self._orig = capture_mod._now
        capture_mod._now = lambda: self.ts
        return self

    def __exit__(self, *a):
        self._mod._now = self._orig


def test_b1_session_2_declaration_reaches_the_reader_and_is_not_discarded(tmp_path):
    """THE B1 REPRO (spec lines 749-761), reproduced RED against the ORIGINAL tree (see the
    implementer's report): session 1 declares only BTC/USD and delivers it; session 2 reopens the
    same capture declaring [BTC/USD, ETH/USD], and ETH/USD is dead from t=0.

    Before B1: `declared_pairs` stayed pinned to session 1's `['BTC/USD']` forever, so the
    reconciliation "passed" by having forgotten ETH/USD was ever asked for, and ETH/USD appeared
    ZERO times in the rendered artifact despite being named in the live recorder's actual
    subscription. After B1: the declared side is the UNION of every session, so ETH/USD's total
    absence is CORRECTLY refused (dead subscription), not silently forgotten.
    """
    p = tmp_path / "b1.jsonl"
    w1 = CaptureWriter(str(p), ["BTC/USD"], "cap-1")
    w1.write_book("BTC/USD", T0, 100.0, 100.2, 1.0, 1.0)
    w1.write_trade("BTC/USD", T0 + 0.5, 100.1, 1.0, "buy")
    w1._fh.close()  # simulated kill — no clean close

    with _FastNow(T0 + 1000.0):
        w2 = CaptureWriter(str(p), ["BTC/USD", "ETH/USD"], "cap-1")
        w2.write_book("BTC/USD", T0 + 1000.5, 100.3, 100.4, 1.0, 1.0)
        w2.write_trade("BTC/USD", T0 + 1001.0, 100.35, 1.0, "buy")
        w2.close(clean=True)

    meta = read_capture(str(p))
    # the reader's declared side now sees BOTH sessions' declarations
    assert meta.declared_pairs == ["BTC/USD", "ETH/USD"], (
        "session 2's declaration must be UNIONED in, not discarded")
    assert meta.observed_pairs == ["BTC/USD"]
    assert not meta.is_complete()
    with pytest.raises(CaptureFormatError, match="DECLARED/OBSERVED pair reconciliation failed"):
        meta.assert_complete()
    # ETH/USD is no longer invisible: it is named, on the record, as a dead declared pair
    assert "ETH/USD" in meta.pair_reconciliation_gaps()["declared_without_data"]
    with pytest.raises(CaptureFormatError, match="ETH/USD"):
        render_report([meta], {}, {})


def test_b1_two_healthy_sessions_reconcile_and_the_second_sessions_pair_renders(tmp_path):
    """Non-vacuity in the other direction: a pair ADDED on restart and received CLEANLY is a pair
    the subscription DID ask for — the mirror direction must not misdiagnose it. If this test
    fails the gate is simply refusing every multi-session capture, proving nothing above."""
    p = tmp_path / "b1_healthy.jsonl"
    w1 = CaptureWriter(str(p), ["BTC/USD"], "cap-1")
    w1.write_book("BTC/USD", T0, 100.0, 100.2, 1.0, 1.0)
    w1.write_trade("BTC/USD", T0 + 0.5, 100.1, 1.0, "buy")
    w1._fh.close()

    # session 2 resumes quickly (10s later, well under the 60s per-jump/edge-hole threshold) and
    # then ticks BOTH pairs densely for a while, so neither pair's own data-bearing span leaves an
    # edge hole against the capture's overall span (B2 item 1) — this test is about B1 alone.
    with _FastNow(T0 + 10.0):
        w2 = CaptureWriter(str(p), ["BTC/USD", "ETH/USD"], "cap-1")
        t = 10.0
        while t <= 70.0:
            w2.write_book("BTC/USD", T0 + t, 100.3, 100.4, 1.0, 1.0)
            w2.write_trade("BTC/USD", T0 + t + 0.01, 100.35, 1.0, "buy")
            w2.write_book("ETH/USD", T0 + t, 50.0, 50.1, 1.0, 1.0)
            w2.write_trade("ETH/USD", T0 + t + 0.01, 50.05, 1.0, "buy")
            t += 1.0
        w2.close(clean=True)

    meta = read_capture(str(p))
    assert meta.declared_pairs == ["BTC/USD", "ETH/USD"]
    assert meta.observed_pairs == ["BTC/USD", "ETH/USD"]
    assert not any(meta.pair_reconciliation_gaps().values())
    assert meta.is_complete()
    meta.assert_complete()
    # the sessions are disclosed, not indistinguishable from a single-session capture
    assert len(meta.sessions) == 2
    assert meta.sessions[0].kind == "start" and meta.sessions[0].pairs == ["BTC/USD"]
    assert meta.sessions[1].kind == "resume"
    assert meta.sessions[1].pairs == ["BTC/USD", "ETH/USD"]
    report = render_report([meta], {}, {})
    assert "sessions: 2 (start, resume)" in report
    assert "HETEROGENEOUS DECLARATIONS" in report

def test_b2_the_7200s_130s_capture_refuses_and_names_the_true_uncovered_fraction(tmp_path):
    """THE B2 FLAGSHIP REPRO (spec lines 712-727). Reproduced RED against the original tree (the
    implementer's report has the verbatim run): a 7200s capture carrying book+trade for its first
    130.5s only, with heartbeats every 10s throughout, gave `uncovered_fraction[BTC/USD] = 0.000%`
    against a true ~98% of the capture un-observed, and `assert_complete()` PASSED.

    After the fix: coverage is denominated in the CAPTURE span (all events, heartbeats included),
    never the data span that collapses together with the numerator — so this now REFUSES, and the
    refusal names the true un-observed fraction, not 0.000%.
    """
    p = tmp_path / "b2.jsonl"
    w = CaptureWriter(str(p), ["BTC/USD"], "cap-b2", channels=["book", "trade"])
    t = 0.0
    while t <= 130.5:
        w.write_book("BTC/USD", T0 + t, 100.0, 100.2, 1.0, 1.0)
        w.write_trade("BTC/USD", T0 + t + 0.01, 100.1, 1.0, "buy")
        t += 5.0
    t = 0.0
    while t <= 7200.0:
        w.write_heartbeat(T0 + t)
        t += 10.0
    w.close(clean=True)

    meta = read_capture(str(p))
    cov = meta.coverage_by_pair["BTC/USD"]
    assert cov.span_s == pytest.approx(7200.0, abs=1.0), "denominated in the CAPTURE span"
    assert cov.uncovered_fraction > 0.95, (
        f"the true un-observed fraction must be reported (~98%), not 0.000% "
        f"(got {cov.uncovered_fraction * 100:.3f}%)")
    assert not meta.is_complete()
    with pytest.raises(CaptureFormatError) as exc:
        meta.assert_complete()
    msg = str(exc.value)
    assert "GATE (iv) REFUSAL" in msg
    # the refusal names the true measured fraction as a NUMBER — not 0.000%. (The message also
    # explicitly CONTRASTS itself with "the 0.000% a data-span-denominated statistic would
    # report", so a bare substring check for "0.000%" would false-positive on that explanatory
    # prose; check instead that the true percentage is actually printed.)
    assert any(pct in msg for pct in ("97.", "98.", "99."))
    assert "BTC/USD: 97." in msg or "BTC/USD: 98." in msg or "BTC/USD: 99." in msg
    with pytest.raises(CaptureFormatError):
        render_report([meta], {}, {})


def test_b2_edge_hole_is_caught_leading_and_trailing_not_just_interior(tmp_path):
    """B2 item 1, isolated: `_find_unrecorded_gaps` used to scan strictly BETWEEN consecutive
    events, so a hole at an EDGE of the timeline (before the first event, or after the last) had
    no consecutive pair spanning it. Anchoring the per-pair scan to the capture's all-event span
    (heartbeats included) catches leading, trailing and both-edge holes the same as an interior
    one."""
    from capture import _find_unrecorded_gaps  # noqa: E402
    # interior hole: already caught pre-B2 (regression pin)
    assert _find_unrecorded_gaps([0.0, 500.0, 1000.0], [], 100.0) != []
    # trailing hole: nothing between consecutive DATA points, but the span extends far past them
    assert _find_unrecorded_gaps([0.0, 10.0], [], 100.0, span=(0.0, 5000.0)) != []
    # leading hole: the data starts long after the span begins
    assert _find_unrecorded_gaps([4990.0, 5000.0], [], 100.0, span=(0.0, 5000.0)) != []
    # both-edge: a single data point in the middle of a huge span
    assert len(_find_unrecorded_gaps([2500.0], [], 100.0, span=(0.0, 5000.0))) == 2
    # no edge hole added when the data already reaches both ends AND stays dense throughout
    # (each individual jump <= threshold, so nothing is flagged at all, edges included)
    assert _find_unrecorded_gaps([0.0, 50.0, 100.0], [], 100.0, span=(0.0, 100.0)) == []


def test_b2_uncovered_fraction_docstring_regression_pins(tmp_path):
    """INVERTED in cycle 2 (run-45) — this test USED to assert `uncovered_fraction == 0.0` for
    `accounted_s <= 0`, i.e. it re-pinned the one value spec B2 item 3 says must never be
    returned there, converting a disclosed deviation into permanently-defended behaviour. It now
    pins the fail-CLOSED direction instead: the NO-CLAIM sentinel (`None`).

    Both halves of the ORIGINAL intent survive and are what this still guards: the fully-gap-
    honest zero-accounted case must NOT be manufactured into a refusal (nothing is hidden there —

                                  recorded_gap_s=0.0, accounted_s=0.0, covered_s=0.0)

    cov_fully_gapped = PairCoverage(pair="X", n_data_events=1, window_s=15.0, span_s=1000.0,
                                     recorded_gap_s=1000.0, accounted_s=0.0, covered_s=0.0)
    assert cov_fully_gapped.uncovered_fraction is None
    # ... and NO-CLAIM is not a bound violation: a fully, honestly disclosed absence still passes.
    meta = CaptureMetadata(path="hand", coverage_by_pair={"X": cov_fully_gapped})
    assert meta.degraded_coverage_pairs() == []
    assert meta.no_claim_coverage_pairs() == ["X"]

def test_b3_the_1500s_capture_still_reports_all_four_pre_registered_horizons(tmp_path):
    """THE B3 REPRO (spec lines 763-776). A 1500s (25-minute) capture in which the pre-registered
    1h markout can never form used to silently OMIT the "1h" row entirely — a reader was shown
    three horizons and never told a fourth was pre-registered and drew nothing. All four now
    appear, always, with the 1h row explicitly zero-sample."""
    p = tmp_path / "b3.jsonl"
    w = CaptureWriter(str(p), ["BTC/USD"], "cap-b3")
    t = 0.0
    px = 100.0
    while t <= 1500.0:
        px += 0.01
        w.write_trade("BTC/USD", T0 + t, px, 1.0, "buy")
        w.write_book("BTC/USD", T0 + t, px, px + 0.1, 1.0, 1.0)
        t += 5.0
    w.close(clean=True)

    meta, results = run_shadow(str(p))
    e1 = compute_estimand1(results["BTC/USD"])
    assert set(e1.by_horizon) == {"1m", "5m", "20m", "1h"}, (
        "all four pre-registered horizons must be present, always")
    assert e1.by_horizon["1h"].n_used == 0
    assert e1.by_horizon["1h"].n_samples == 0
    assert e1.by_horizon["1h"].is_evaluable is False
    assert e1.by_horizon["1m"].n_used > 0, "the other horizons really did draw samples"

    report = render_report([meta], {"BTC/USD": e1}, {})
    assert "| 1h |" in report, "the 1h row must be a STATED row in the rendered table"
    assert "(MISSING" not in report, (
        "compute_estimand1 always populates all four horizons now — the MISSING fallback text "
        "is a render-level backstop for a hand-built result and should not fire here")

def test_overrefusal_major_a_book_only_pair_is_disclosed_not_refused(tmp_path):
    """THE OVER-REFUSAL REPRO (spec lines 781-795). A healthy 3-pair capture where SOL/USD has a
    live, densely-covered book and ZERO prints for the whole capture used to refuse the WHOLE
    render — discarding every number from BTC/USD and ETH/USD too — over a pair the gate could
    plainly see (a coverage record, a cadence row, book-derived mids). Missing `trade` on a pair
    whose `book` is alive is a market fact (no prints in the window), not a broken subscription;
    missing `book` on an observed pair still refuses.
    """
    p = tmp_path / "overref.jsonl"
    w = CaptureWriter(str(p), ["BTC/USD", "ETH/USD", "SOL/USD"], "cap-or",
                       channels=["book", "trade"])
    t = 0.0
    while t <= 600.0:
        w.write_book("BTC/USD", T0 + t, 100.0, 100.1, 1.0, 1.0)
        w.write_trade("BTC/USD", T0 + t + 0.01, 100.05, 1.0, "buy")
        w.write_book("ETH/USD", T0 + t, 50.0, 50.1, 1.0, 1.0)
        w.write_trade("ETH/USD", T0 + t + 0.01, 50.05, 1.0, "buy")
        w.write_book("SOL/USD", T0 + t, 10.0, 10.05, 1.0, 1.0)  # book only, never a print
        t += 5.0
    w.close(clean=True)

    meta = read_capture(str(p))
    assert meta.observed_pair_channels["SOL/USD"] == ["book"]
    assert "SOL/USD" in meta.coverage_by_pair, "the gate can plainly see this pair"
    assert meta.disclosed_empty_trade_cells() == ["SOL/USD"]
    assert meta.pair_reconciliation_gaps()["declared_channel_without_data"] == [], (
        "the missing trade cell must NOT gate — it is disclosed instead")
    assert meta.is_complete()
    meta.assert_complete()  # must not raise

    meta2, results = run_shadow(str(p))
    e1 = {pr: compute_estimand1(r) for pr, r in results.items()}
    e2 = {pr: compute_estimand2(r) for pr, r in results.items()}
    report = render_report([meta2], e1, e2)
    assert "disclosed empty trade cell(s)" in report
    assert "['SOL/USD']" in report
    assert "## BTC/USD" in report and "## ETH/USD" in report, (
        "the good pairs' numbers must survive — nothing is discarded")


def test_overrefusal_mirror_a_missing_book_cell_still_refuses(tmp_path):
    """The asymmetry is deliberate: an actively-printing pair (trade present) missing its `book`
    cell entirely is still a broken subscription, not a market fact, and still REFUSES."""
    p = tmp_path / "missing_book.jsonl"

    meta = read_capture(str(p))
    assert meta.disclosed_empty_trade_cells() == [], "this is the BOOK-missing direction"

class _AckWS:
    def __init__(self, items):
        self._items = list(items)

    async def send(self, msg):
        pass

    async def recv(self):
        if not self._items:
            await asyncio.sleep(999)
        return self._items.pop(0)


class _AckCM:
    def __init__(self, ws):
        self._ws = ws

    async def __aenter__(self):
        return self._ws

    async def __aexit__(self, *a):
        return False


async def _fast_sleep(_s):
    return None

    return json.dumps({"method": "subscribe",

    """Spec B2 item 3 is unconditional: "that branch is fail-CLOSED ... `0.0` is the one value
    that must never be returned there." Cycle 1 left the branch byte-identical to HEAD, argued in
    a rewritten comment that it was safe, and ADDED a test re-pinning the forbidden value.

    The sentinel (`None` — "no claim made") is chosen over `1.0` deliberately: the requirement is
    that the branch cannot BACK a 0.0, not that it must claim total absence, and `1.0` manufactures
    a refusal for captures with nothing hidden.

    RED against cycle 1::

        assert cov_fully_gapped.uncovered_fraction is None, ...
        AssertionError: `accounted_s <= 0` makes NO coverage claim; 0.0 is the one value that
        must never be returned there
        assert 0.0 is None
    """
    cov_zero_span = PairCoverage(pair="X", n_data_events=1, window_s=15.0, span_s=0.0,
                                  recorded_gap_s=0.0, accounted_s=0.0, covered_s=0.0)
    assert cov_zero_span.uncovered_fraction is None
    cov_fully_gapped = PairCoverage(pair="X", n_data_events=1, window_s=15.0, span_s=1000.0,
                                     recorded_gap_s=1000.0, accounted_s=0.0, covered_s=0.0)
    assert cov_fully_gapped.uncovered_fraction is None, (
        "`accounted_s <= 0` makes NO coverage claim; 0.0 is the one value that must never be "
        "returned there")

    meta = CaptureMetadata(path="hand", coverage_by_pair={"X": cov_fully_gapped})
    assert meta.no_claim_coverage_pairs() == ["X"]

# --- MAJOR: the "abort loudly" that was an unbounded reconnect storm -------------------------------

def test_cycle2_a_recurring_subscribe_ack_mismatch_reaches_a_terminal_state(tmp_path):
    """Defect #2's abort used `_GapAlreadyRecorded` — the idiom for a TRANSIENT failure — so
    `run()`'s loop backed off to `backoff_max_s` (60s) and retried forever. For a STATIC
    misconfiguration (Kraken's legacy `XBT/USD` vs a config declaring `BTC/USD`) every reconnect
    re-subscribes identically and reconciles false again: an unbounded ~60s reconnect storm
    against the venue for the life of a multi-day process, with no terminal signal. The only
    coverage ran at `max_iterations=1`, which cannot see a storm.

    RED against cycle 1 (`max_iterations=5`)::

        with pytest.raises(SubscriptionConfigMismatch):
        E   ImportError: cannot import name 'SubscriptionConfigMismatch' from 'recorder'
        # and, with the import removed:
        assert rec.attempts <= 3, ...
        AssertionError: a STATIC config mismatch must terminate, not storm the venue: 5 connection
        attempts, every one reconciling false identically, with no terminal signal
        assert 5 <= 3
    """
    from recorder import SUBSCRIBE_ACK_MISMATCH_TERMINAL_ATTEMPTS, SubscriptionConfigMismatch

    out = str(tmp_path / "ackstorm.jsonl")

    def _fresh_ws(_url):
        # every reconnect re-subscribes and the venue echoes the SAME legacy spelling back
        return _AckCM(_AckWS([_ack_frame("book", "XBT/USD"), _ack_frame("trade", "BTC/USD"),
                              _heartbeat_frame()]))

    cfg = RecorderConfig(pairs=["BTC/USD"], out_path=out, capture_id="ackstorm",
                          heartbeat_timeout_s=5.0, backoff_base_s=0.001, backoff_max_s=0.01)
    rec = LiveRecorder(cfg, _fresh_ws, sleep_fn=_fast_sleep)
    with pytest.raises(SubscriptionConfigMismatch):
        asyncio.run(rec.run(max_iterations=5))
    rec.close(clean=False)

    assert rec.attempts <= SUBSCRIBE_ACK_MISMATCH_TERMINAL_ATTEMPTS, (
        f"a STATIC config mismatch must terminate, not storm the venue: {rec.attempts} "
        f"connection attempts, every one reconciling false identically")
    lines = [json.loads(l) for l in open(out) if l.strip()]
    gaps = [o for o in lines if o.get("type") == "gap"]
    assert any("TERMINAL" in g.get("detail", "") for g in gaps), (
        "the terminal outcome must be written into the capture so it reaches the artifact")
    assert all(g["reason"] in GAP_REASONS for g in gaps)


def test_cycle2_a_transient_ack_mismatch_still_only_reconnects(tmp_path):
    """The DISTINCTION is the fix, not the abort: a mismatch that does NOT recur (the venue's
    first ack raced, the second connection acks correctly) must still be a reconnect, never a
    terminal kill of a multi-day capture."""
    from recorder import SubscriptionConfigMismatch

    out = str(tmp_path / "acktransient.jsonl")
    state = {"n": 0}

    def _ws(_url):
        state["n"] += 1
        if state["n"] == 1:
            frames = [_ack_frame("book", "XBT/USD"), _heartbeat_frame()]
        else:
            frames = [_ack_frame("book", "BTC/USD"), _heartbeat_frame(), _trade_ok_frame()]
        return _AckCM(_AckWS(frames))

    cfg = RecorderConfig(pairs=["BTC/USD"], out_path=out, capture_id="acktrans",
                          heartbeat_timeout_s=5.0, backoff_base_s=0.001, backoff_max_s=0.01)
    rec = LiveRecorder(cfg, _ws, sleep_fn=_fast_sleep)
    asyncio.run(rec.run(max_iterations=3))  # must NOT raise
    rec.close(clean=False)
    lines = [json.loads(l) for l in open(out) if l.strip()]
    assert not any("TERMINAL" in o.get("detail", "") for o in lines if o.get("type") == "gap")

# CYCLE 3 (run 45) — THE BLOCKER: an honest market fact must not be an instrument-defect refusal
#
# Cycle 2 correctly stopped counting keepalive-frozen book time as observation, then routed the
# result into `PairCoverage.uncovered_fraction` -> `degraded_coverage_pairs()` ->
# `is_complete()`/`assert_complete()` -> `render._assert_all_captures_complete`: a WHOLE-CAPTURE
# gate that runs before any number is built and is independent of `--pairs`.
#
# RED against cycle 2, on the fixture below (production `CaptureWriter`, 3 pairs, BTC/USD and
# ETH/USD perfectly healthy, SOL/USD's real top of book frozen for ONE stretch)::
#
#     SOL/USD uncovered_fraction=0.02720  stale_book_s=1200.0
#     degraded: ['SOL/USD']
#     assert_complete(): GATE (iv) REFUSAL
#     rc (no --pairs)              = 1, artifact exists: False
#     rc (--pairs BTC/USD,ETH/USD) = 1, artifact exists: False   <-- no escape hatch
#
# The whole multi-day report is discarded, both healthy pairs with it, and the refusal's own
# remedies ("re-cut the capture", "re-run the recorder") are unusable on data already recorded.
# The fix applies the asymmetry this spec already established for the empty trade cell (in-gate
# defect #1): an INSTRUMENT DEFECT refuses, a MARKET FACT is disclosed. Genuine data absence —
# the throttling subscription of `test_coverage_aggregate.py:185` — still refuses whole-capture,
# unchanged and deliberately so.
# =================================================================================================

CYCLE3_KEEPALIVE_S = 5.0


def _thin_pair_capture(path, span_s=3600.0, stale=(("SOL/USD", 1000.0, 1120.0),),
                       pairs=("BTC/USD", "ETH/USD", "SOL/USD"), channels=("book", "trade")):
    """A healthy multi-pair capture through the PRODUCTION writer in which the named pairs' real
    top of book sits still over the given stretches, with the recorder emitting on its declared
    `book_keepalive_s` floor throughout — a weekend/overnight lull on a thin pair. No prints
    arrive over a frozen stretch either, which is what makes it a lull rather than a bug."""
    w = CaptureWriter(str(path), list(pairs), "capThin", channels=list(channels),
                       book_emission_policy="change_or_keepalive",
                       book_keepalive_s=CYCLE3_KEEPALIVE_S)
    px = {p: 100.0 + 10.0 * i for i, p in enumerate(pairs)}
    frozen_by_pair = {}
    for pair, lo, hi in stale:
        frozen_by_pair.setdefault(pair, []).append((lo, hi))
    t = 0.0
    while t <= span_s:
        for p in pairs:
            is_frozen = any(lo <= t < hi for lo, hi in frozen_by_pair.get(p, ()))
            if not is_frozen:
                px[p] += 0.01
            w.write_book(p, T0 + t, px[p], px[p] + 0.1, 1.0, 1.0)
            if not is_frozen and "trade" in channels:
                w.write_trade(p, T0 + t + 0.01, px[p] + 0.05, 1.0, "buy")
        if t % 10.0 == 0.0:
            w.write_heartbeat(T0 + t)
        t += CYCLE3_KEEPALIVE_S
    w.close(clean=True)
    return str(path)


def test_cycle3_a_thin_pairs_frozen_book_no_longer_discards_the_healthy_pairs_beside_it(tmp_path):
    """THE BLOCKER, closed. SOL/USD is excluded BY NAME and every other pair still renders."""
    path = _thin_pair_capture(tmp_path / "thin.jsonl")
    meta = read_capture(path)

    cov = meta.coverage_by_pair["SOL/USD"]
    assert cov.stale_book_s == pytest.approx(120.0, abs=0.01), "detection is unchanged"
    assert cov.uncovered_fraction > meta.max_uncovered_fraction, (
        "the honest un-observed fraction is over the bound — that measurement is preserved")
    assert cov.uncovered_fraction_counting_stale == pytest.approx(0.0, abs=1e-6), (
        "and NOTHING was missing beside the frozen book: with the keepalives counted back in the "
        "pair is inside the bound, which is exactly what separates it from a throttling "
        "subscription")
    assert meta.degraded_coverage_pairs() == []
    assert meta.stale_book_excluded_pairs() == ["SOL/USD"]
    for healthy in ("BTC/USD", "ETH/USD"):
        assert meta.coverage_by_pair[healthy].uncovered_fraction == pytest.approx(0.0, abs=1e-6)
    meta.assert_complete()

    out = tmp_path / "art.md"
    assert cli.main(["--capture", path, "--out", str(out)]) == 0
    report = out.read_text()
    assert "## BTC/USD" in report and "## ETH/USD" in report, (
        "the healthy pairs' numbers must survive a frozen book on a pair beside them")
    assert "## SOL/USD" not in report, "an excluded pair backs no estimand section"
    # ... and the exclusion is impossible to miss, and cannot be read as "never subscribed".
    assert "PAIRS EXCLUDED FROM THIS REPORT" in report
    assert "were subscribed, were received, and are NOT reported" in report
    assert "minute-grid claim is COMPROMISED" in report
    assert report.index("PAIRS EXCLUDED FROM THIS REPORT") < report.index("## BTC/USD")

def test_cycle3_the_degraded_remedy_is_split_and_every_branch_is_achievable(tmp_path):
    """ITEM 2. "Re-cut the capture around the degraded stretch" is VACUOUS when the whole capture
    is the stretch. The remedy splits on a measured quantity — the longest contiguous observed
    run — so a pair with a healthy stretch to cut to is told to cut to it, and a pair without one
    is told plainly that no reader-side action on this file can produce a number for it."""
    # Both degraded pairs THROTTLE rather than go dark — a 55s cadence clears the 60s per-jump
    # threshold, which is the only way to reach the aggregate gate at all (a pair that simply
    # stops is caught one check earlier, by the per-pair data-gap scan).
    p = tmp_path / "remedy.jsonl"
    w = CaptureWriter(str(p), ["BTC/USD", "ETH/USD", "SOL/USD"], "capRem", channels=["book"])
    px = 100.0
    t = 0.0
    while t <= 3600.0:
        px += 0.01
        w.write_book("BTC/USD", T0 + t, px, px + 0.1, 1.0, 1.0)   # dense throughout: the anchor
        if t < 1800.0 or (t % 55.0) < 5.0:
            w.write_book("SOL/USD", T0 + t, px, px + 0.1, 1.0, 1.0)  # dense half, throttled half
        if (t % 55.0) < 5.0:
            w.write_book("ETH/USD", T0 + t, px, px + 0.1, 1.0, 1.0)  # throttled for its whole life
        t += 5.0
    w.close(clean=True)
    meta = read_capture(str(p))
    assert meta.coverage_by_pair["SOL/USD"].longest_observed_run_s >= 60.0
    assert meta.coverage_by_pair["ETH/USD"].longest_observed_run_s < 60.0
    assert set(meta.degraded_coverage_pairs()) == {"ETH/USD", "SOL/USD"}
    with pytest.raises(CaptureFormatError) as exc:
        meta.assert_complete()
    msg = str(exc.value)
    assert "Remedy split by failure kind" in msg
    assert "DO still have a healthy stretch inside this file" in msg and "SOL/USD" in msg
    assert "re-cutting is VACUOUS" in msg and "['ETH/USD']" in msg
    assert "record a FRESH capture" in msg
    assert "Re-cut the capture around the degraded stretch" not in msg, (
        "the vacuous single remedy is gone")
