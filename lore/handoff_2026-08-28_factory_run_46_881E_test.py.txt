def _mk_ckpt(root, name, mtime=None):
    p = os.path.join(root, name)
    os.makedirs(p, exist_ok=True)
    (open(os.path.join(p, "checkpoint"), "w")).close()
    if mtime is not None:
        os.utime(p, (mtime, mtime))
    return p


def _run_gc(args, cwd):
    return subprocess.run([sys.executable, GC_PATH] + args, cwd=cwd,
                          capture_output=True, text=True)


def _d64_hostile_tree(tmp_path):
    """D64's OWN hostile scenario (mirrors test_ckpt_881_protection_model.py's own
    `_d64_hostile_tree`, rebuilt here so this file is self-contained): a run dir whose only
    coverage comes from a `covers:` line in a totally unrelated experiment id."""
    runs = tmp_path / "runs"
    for i in (1, 2, 3):
        _mk_ckpt(str(runs / "target_run"), f"checkpoint_{i}")
    hostile = tmp_path / "experiments" / "hostile_unrelated"
    hostile.mkdir(parents=True)
    (hostile / "PROTECTED.txt").write_text("covers: runs/target_run\n")
    return runs / "target_run"

def _survivors(run_dir):
    return sorted(p.name for p in run_dir.iterdir())


# TASK_881D: `--apply` now implies `--write-index` (an auditable `checkpoints_index.json` per
# run dir) and appends `prunes.jsonl` — sanctioned artifacts a DELETING run writes into run_dir
# itself, not checkpoints, so a survivors comparison after an actual delete must exclude them
# (same convention as test_ckpt_881i_manifest_ingestion.py's own `_AUDIT_FILES`). A REFUSED run
# dir gets neither (TASK_881 BLOCKER A: an unresolved run dir gets no index write at all), so the
# refusal-case assertions elsewhere in this file are unaffected and stay bare.
_AUDIT_FILES = {"checkpoints_index.json", "prunes.jsonl"}


def _survivor_checkpoints(run_dir):
    return [n for n in _survivors(run_dir) if n not in _AUDIT_FILES]

def test_resolve_bound_coverage_can_only_remove_never_add(tmp_path):
    """A direct, constructive proof of the non-negotiable: build a tree where the base predicate
    covers 2 run dirs (one correlated, one not) and confirm the bound predicate's covered set is
    a SUBSET of the base predicate's — never a superset, never a disjoint addition."""
    runs = tmp_path / "runs"
    _mk_ckpt(str(runs / "832_a"), "checkpoint_1")
    _mk_ckpt(str(runs / "unrelated_b"), "checkpoint_1")
    exp = tmp_path / "experiments" / "832_owner"
    exp.mkdir(parents=True)
    (exp / "PROTECTED.txt").write_text(
       f"covers: {runs / '832_a'}\ncovers: {runs / 'unrelated_b'}\n")

    manifests = gc.load_manifests([str(tmp_path / "experiments")])
    named = manifests.protect | manifests.covers
    run_dirs = [str(runs / "832_a"), str(runs / "unrelated_b")]

    base = gc.resolve_coverage(run_dirs, named, set(manifests.dangling))
    bound = gc.resolve_bound_coverage(run_dirs, named, set(manifests.dangling),
                                      citations=manifests.citations,
                                      scan_dirs=[str(tmp_path / "experiments")],
                                      roots=[str(runs)])

    assert base[str(runs / "832_a")] is True
    assert base[str(runs / "unrelated_b")] is True
    assert bound[str(runs / "832_a")] is True, "the correlated grant must survive the binding"
    assert bound[str(runs / "unrelated_b")] is False, "the uncorrelated grant must be removed"
    for rd in run_dirs:
        assert (not base[rd]) or (bound[rd] in (True, False))
        assert bound[rd] <= base[rd], f"{rd}: bound coverage exceeded base coverage"

def test_authority_correlation_is_scoped_to_the_run_dirs_own_owning_segment():
    """The cycle-2 BLOCKER at the unit level (the CLI-level contract lives in the section at the
    end of this file). `_run_dir_task_number` is the ONE quantity the decision reads, and it is
    computed from the run dir's path RELATIVE to a configured root."""
    # (a) an inner, non-owning segment confers nothing
    assert gc._run_dir_task_number("runs/47_task/50_epoch", _ROOTS) is None
    assert not gc._authority_correlated("50_bogus", "runs/47_task/50_epoch", _ROOTS)
    # (b) a shared numeric container confers nothing over another task's child
    assert gc._run_dir_task_number("runs/000_smoke/769_value_clip", _ROOTS) is None
    assert not gc._authority_correlated("000_ci", "runs/000_smoke/769_value_clip", _ROOTS)
    # (c) ancestors ABOVE the roots are not the task namespace
    assert gc._run_dir_task_number("/mnt/2tb_disk/runs/target", ["/mnt/2tb_disk/runs"]) is None
    assert not gc._authority_correlated("2_bogus", "/mnt/2tb_disk/runs/target",
                                        ["/mnt/2tb_disk/runs"])
    # (d) the real corpus shapes still designate their owner
    assert gc._run_dir_task_number("runs/832_e5", _ROOTS) == "832"
    assert gc._run_dir_task_number("runs/875_wf/f1_s0", _ROOTS) == "875"
    assert gc._authority_correlated("875_wf_generalization", "runs/875_wf/f1_s0", _ROOTS)
    # (e) fail-closed: unrooted, and roots omitted entirely
    assert gc._run_dir_task_number("/elsewhere/832_e5", _ROOTS) is None
    assert not gc._authority_correlated("832_linear_imprint", "/elsewhere/832_e5", _ROOTS)
    assert not gc._authority_correlated("832_linear_imprint", "runs/832_e5", [])

    """The binding ACTIVE (citations+scan_dirs given) but `roots` omitted must fail CLOSED. It is
    keyword-optional only so a caller omission can never raise inside the in-run pruner's
    apply=True path (a raise there flips train_jax_fast.py's exit code) — the cost of that is
    paid in refusals, never in grants."""
    runs = tmp_path / "runs"
    _mk_ckpt(str(runs / "832_a"), "checkpoint_1")
    exp = tmp_path / "experiments" / "832_owner"
    exp.mkdir(parents=True)
    (exp / "PROTECTED.txt").write_text(f"covers: {runs / '832_a'}\n")
    manifests = gc.load_manifests([str(tmp_path / "experiments")])

    assert gc.resolve_bound_coverage(rd, named, set(manifests.dangling), **kw)[rd[0]] is False
    assert gc.resolve_bound_coverage(rd, named, set(manifests.dangling), roots=[str(runs)],

# ======================================================== cycle-2 BLOCKER: the OWNING segment only
# D64's amendment binds coverage to "the run dir's task/experiment segment" — SINGULAR, the run
# dir's OWN owning id. Cycle 1 implemented it as "any segment of the run dir's full absolute
# realpath", which grants standing to (a) inner segments that are not task ids at all
# (`50_epoch` = epoch 50), (b) an unrelated sibling's number inside a shared scratch CONTAINER
# (`runs/000_smoke/769_value_clip` — 30 unrelated tasks under one `000_` parent, live in this
# repo today), and (c) ancestors ABOVE the configured roots that are not part of the repo's
# namespace at all (`/mnt/2tb_disk/...` -> "2"). Each of those is an authority the ruling never
# granted, on an IRREVERSIBLE deleter. These five tests are the cycle-2 fix's contract.


def _covers_tree(tmp_path, *, run_rel: str, manifest_id: str, roots_dir: str = "runs",
                 cwd_sub: str = ""):
    """A minimal tree: one checkpoint-bearing run dir at `run_rel`, one manifest at
    `experiments/<manifest_id>/PROTECTED.txt` whose `covers:` line names it. Returns the cwd the
    CLI should run from (so `--roots`/`--experiments-dir` stay RELATIVE, as production uses
    them) and the run dir path as the tool will print it."""
    base = tmp_path / cwd_sub if cwd_sub else tmp_path
    run_dir = base / run_rel
    _mk_ckpt(str(run_dir), "checkpoint_1")
    exp = base / "experiments" / manifest_id
    exp.mkdir(parents=True)
    (exp / "PROTECTED.txt").write_text(f"covers: {run_rel}\n")
    return str(base), run_rel


def _refused(tmp_path, **kw):
    cwd, run_rel = _covers_tree(tmp_path, **kw)
    r = _run_gc(["--roots", kw.get("roots_dir", "runs"), "--experiments-dir", "experiments",
                 "--min-free-gb", "1e18"], cwd=cwd)
    assert r.returncode == 0, r.stdout + r.stderr
    return f"REFUSE run dir {run_rel}" in r.stdout, r.stdout


def test_inner_segment_number_does_not_confer_standing(tmp_path):
    """(1) `experiments/50_bogus` must NOT cover `runs/47_task/50_epoch`: the run dir's owning
    segment is `47_task`; `50_epoch` is an EPOCH counter that happens to be digit-led. Under the
    cycle-1 predicate `_path_task_numbers` returned {'47','50'} and plain set membership let
    `50_bogus` through."""
    refused, out = _refused(tmp_path, run_rel="runs/47_task/50_epoch", manifest_id="50_bogus")
    assert refused, (
        "an UNRELATED manifest whose id matches an inner, non-owning path segment was granted "
        f"standing over the run dir:\n{out}")


def test_shared_numeric_container_does_not_confer_standing_over_its_children(tmp_path):
    """(2) The LIVE repo shape: `runs/000_smoke/` is a shared scratch container holding 30
    entries belonging to unrelated tasks (`753_step7`, `769_value_clip`, `771_chain_bugfix`, ...).
    A future `experiments/000_*/PROTECTED.txt` must NOT thereby gain standing over all thirty.

def test_ancestor_above_the_roots_does_not_confer_standing(tmp_path):
    """(3) Ancestor pollution. The run dir lives under a numerically-named directory that is NOT
